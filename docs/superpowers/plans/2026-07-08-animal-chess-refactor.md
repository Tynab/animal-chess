**Goal:** Aggressively refactor the Animal Chess Python source (dedup, table-driven metadata, extracted encoder/evaluator) and add maximum-density Vietnamese comments, without changing any on-disk format — then rewrite the repo's docs/config files.

**Architecture:** A golden/smoke harness captured from the *current* code (`tools/smoke_test.py` + `tools/golden.json`) is the regression oracle. ## Global Constraints

- **Platform:** Windows / PowerShell; commands use `py`. The smoke harness runs headless — it builds `Board(is_copy=True)` (image-free), so it needs **no display, no assets, no LFS**.
- **Byte-identical formats invariant (NEVER change output):** `Log.board_to_enum` (63-char column-major, case-encoded), `encode_board` tensor values, `PIECE_ENCODING` (dark `+1..+8`, light `-1..-8`, `-`=0), move notation `A1B2` (`move_to_enum`/`enum_to_position`), `make_move`/`undo_move` semantics + `captured_pieces`↔`move_history` sync, and `Log.COLUMNS`.
- **Preserve bot move-selection behavior** (minimax/alpha-beta coin flip, depth 2, BFS/A* routing, `algorithm_random` weighting). Refactor internal code shape only.
- **One real class per piece type** (`Rat`, `Cat`, …, `Elephant`) — required for `isinstance` checks in Rat/Elephant rules. Only metadata is table-sourced.
- **Comments:** maximum-density **Vietnamese**, accurate to the code (carry the audit corrections). Applies to `main.py`, `scripts/**/*.py`, and notebook **code cells** only — never output cells, assets, CSV, or the `.h5`.
- **Keep public import surfaces** that other modules/notebooks depend on (e.g. `GameManager`, `Board`, `Log.board_to_enum`, `Log.enum_to_postion` alias).
- **Per-task gate:** `py tools\smoke_test.py` prints `SMOKE_OK`; for any file that changed, `py -m compileall -q <files>` and `py -m ruff check <files>` are clean. Commit after each task.

```python
# tools/smoke_test.py
"""Bộ smoke/golden test cho Animal Chess.

Chạy headless hoàn toàn: dùng ``Board(is_copy=True)`` để dựng bàn logic không
nạp ảnh, nên không cần display, asset hay Git LFS. Dùng làm lưới chống hồi quy
cho đợt refactor: giá trị vàng được chốt từ code GỐC rồi kiểm lại sau mỗi bước.
"""
import json
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(ROOT))

from scripts.board import Board
from scripts.bot import Bot
from scripts.common import PlayerSide
from scripts.log import Log

# encode_board chuyển sang scripts.encoding ở Task 2; import mềm để test không vỡ.
try:
    from scripts.encoding import encode_board
except ImportError:
    from scripts.manager import GameManager
    encode_board = GameManager.encode_board

GOLDEN_PATH = Path(__file__).resolve().parent / "golden.json"


def collect() -> dict:
    """Thu thập các bất biến format/luật/bot từ bàn khởi đầu."""
    board = Board(is_copy=True)                     # bàn đầu ván, không ảnh
    initial = Log.board_to_enum(board)
    tensor = encode_board(initial)

    dark_moves = len(board.get_valid_moves(PlayerSide.DARK))
    light_moves = len(board.get_valid_moves(PlayerSide.LIGHT))

    first = board.get_valid_moves(PlayerSide.LIGHT)[0]   # nước hợp lệ đầu tiên
    board.make_move(first)
    after = Log.board_to_enum(board)
    board.undo_move(first)
    restored = Log.board_to_enum(board)

    fresh = Board(is_copy=True)
    score, best = Bot.minimax(fresh, PlayerSide.LIGHT, 2, True)

    return {
        "initial_board": initial,
        "tensor_shape": list(tensor.shape),
        "tensor_flat": [float(v) for v in tensor.reshape(-1)],
        "dark_moves": dark_moves,
        "light_moves": light_moves,
        "after_first_move": after,
        "restored_after_undo": restored,
        "minimax_score": score,
        "minimax_best_count": len(best),
    }


def main() -> int:
    data = collect()
    if "--capture" in sys.argv:
        GOLDEN_PATH.write_text(json.dumps(data, indent=2), encoding="utf-8")
        print(f"Đã ghi golden: {GOLDEN_PATH}")
        return 0
    golden = json.loads(GOLDEN_PATH.read_text(encoding="utf-8"))
    ok = True
    for key, want in golden.items():
        if data[key] != want:
            ok = False
            print(f"[SAI] {key}: want {want!r} got {data[key]!r}")
    if data["restored_after_undo"] != data["initial_board"]:
        ok = False
        print("[SAI] undo không khôi phục bàn ban đầu")
    print("SMOKE_OK" if ok else "SMOKE_FAIL")
    return 0 if ok else 1


if __name__ == "__main__":
    raise SystemExit(main())
```

- [ ] **Step 1: Create `scripts/encoding.py`** — move the exact bodies (verbatim math) with max-density VN comments:

```python
"""Mã hóa bàn cờ sang tensor cho model CNN.

Tách khỏi ``manager.py``/``train.ipynb`` để chỉ còn MỘT nguồn sự thật cho phép
mã hóa. Định dạng phải giữ nguyên byte-for-byte vì ``models/best_model.h5`` đã
được huấn luyện theo đúng layout này.
"""
import numpy as np

# Quân phe tối là số dương 1..8, phe sáng là số âm -1..-8, ô trống là 0.
PIECE_ENCODING = {
    "-": 0, "r": 1, "c": 2, "d": 3, "w": 4, "p": 5, "t": 6, "l": 7, "e": 8,
    "R": -1, "C": -2, "D": -3, "W": -4, "P": -5, "T": -6, "L": -7, "E": -8,
}


def encode_piece(piece_char: str) -> int:
    """Trả mã số của một ký tự quân; ký tự lạ coi như ô trống (0)."""
    return PIECE_ENCODING.get(piece_char, 0)


def encode_board(board_str: str) -> np.ndarray:
    """Chuỗi 63 ô -> tensor (9,7) đúng hướng tập dữ liệu/model cũ."""
    if len(board_str) != 63:
        raise ValueError(f"Bàn cờ phải có 63 ký tự, nhận {len(board_str)}.")
    # reversed -> reshape(7,9) -> transpose -> flip cả hai trục: phép xoay/lật
    # cố định để khớp hướng bàn của tập dữ liệu gốc. KHÔNG đổi.
    values = np.fromiter(
        (encode_piece(char) for char in reversed(board_str)),
        dtype=np.float32, count=63,
    )
    return np.flip(values.reshape(7, 9).T, axis=(0, 1)).copy()
```
